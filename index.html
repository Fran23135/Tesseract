"block-config": { "enabled": bool(bc.get("enabled", False)), "pb_start": bc.get("pb_start", "begin"), "pb_end": bc.get("pb_end", "end"), }, "transform-code": { "enabled": bool(tc.get("enabled", """Convierte el JSON real del disco (claves con guion medio) al dict interno (guion bajo) que usa el formulario de la UI.""" cfg = get_default_config() sm = data.get("strict-mode", {}) or {} typing = sm.get("typing", {}) or {} bc = data.get("block-config", {}) or {} tc = data.get("transform-code", {}) or {} cfg["strict_mode"]["enabled"] = bool(sm.get("enabled", cfg["strict_mode"]["enabled"])) """Escribe cfg (dict interno, guion bajo) como JSON real en config.tsc.""" cfg_dir = get_config_dir() os.makedirs(cfg_dir, exist_ok=True) tsc_path = get_config_path() disk_json = to_disk_json(cfg) with open(tsc_path, "w", encoding="utf-8") as f: json.dump(disk_json, f, indent=2, ensure_ascii=False) f.write("\n") return tsc_path @app.on_event("startup") async def reset_config_on_startup(): """Al Si no existe, NO lo crea: solo devuelve valores de referencia en modo libre para rellenar el formulario, junto con _exists=False.""" tsc_path = get_config_path() if os.path.exists(tsc_path): try: with open(tsc_path, "r", encoding="utf-8") as f: data = json.load(f) cfg = from_disk_json(data) cfg["_exists"] = True cfg["_path"] = tsc_path return JSONResponse(cfg) except Exception as e: return JSONResponse( {"error": f"config.tsc existe pero no se pudo leer como JSON: {e}"}, status_code=500, ) cfg = get_default_config() cfg["_exists"] = False cfg["_path"] = tsc_path return JSONResponse(cfg) @app.post("/config") async def write_config(payload: dict = Body(...)): """Endpoint manual para inspección/pruebas puntuales. OJO: el flujo normal del playground YA NO depende de este endpoint para persistir cambios — cada ejecución manda su propia configuración por WebSocket, se aplica solo para esa ejecución, y el entorno se resetea a modo libre justo después. Así ningún usuario se queda con la configuración de otro.""" bc = payload.get("block_config", {}) or {} for key in ("pb_start", "pb_end"): v = bc.get(key) if v is not None and len(str(v)) > PB_DELIMITER_MAX_LEN: return JSONResponse( {"error": f"'{key}' supera el límite de {PB_DELIMITER_MAX_LEN} caracteres."}, status_code=400, ) tsc_path = write_config_to_disk(payload) return JSONResponse({"status": "ok", "path": tsc_path}) # Lock para evitar que dos ejecuciones simultáneas se pisen al escribir # config.tsc justo antes de lanzar el intérprete. config_write_lock = asyncio.Lock() @app.websocket("/ws") async def websocket_endpoint(websocket: WebSocket): await websocket.accept() process = None output_task = None ws_id = id(websocket) tmp_dir = f"/tmp/tess_{ws_id}" try: payload = await websocket.receive_json() code = payload.get("code", "") syntax = payload.get("syntax", "-b") target_val = payload.get("target", "interp") is_debug = payload.get("debug", False) # ===================================================================== # ENTORNO (config.tsc) DE ESTA EJECUCIÓN # ===================================================================== # Cada ejecución trae su propia configuración de entorno (o modo libre # si el usuario no tocó el panel de configuración). Se escribe justo # antes de correr, y se resetea a modo libre en el "finally" de abajo # apenas termina — así ningún usuario hereda la configuración guardada # por otro. env_cfg = payload.get("config") or get_default_config() async with config_write_lock: try: write_config_to_disk(env_cfg) except Exception as e: print(f"[config.tsc] No se pudo aplicar la configuración de esta ejecución: {e}") os.makedirs(tmp_dir, exist_ok=True) source_file = os.path.join(tmp_dir, "main.tss") with open(source_file, "w", encoding="utf-8") as f: f.write(code) interpreter_path = os.path.abspath("./interpreter") cmd = [] # ===================================================================== # FASE 1: DEJAR TRABAJAR AL COMPILADOR/TRANSPILADOR (WEB Y NATIVE) # ===================================================================== if target_val in ["web", "native"]: comp_cmd = [interpreter_path, source_file, syntax, "--target", target_val] if is_debug: comp_cmd.append("-d") # Lanzamos el proceso de compilación puramente asíncrono comp_process = await asyncio.create_subprocess_exec( *comp_cmd, stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.STDOUT, cwd=tmp_dir ) # CORRECCIÓN: Leer por fragmentos crudos (chunks) para no bloquearse while True: chunk = await comp_process.stdout.read(1024) if not chunk: break await websocket.send_json({"type": "stdout", "data": chunk.decode("utf-8", errors="replace")}) await comp_process.wait() return_code = comp_process.returncode if return_code != 0: await websocket.send_json({"type": "exit", "data": f"\n[Compilación finalizada con código de error {return_code}]"}) await websocket.close() return # ================================================================= # FASE 1.5: BUSCAR Y PREPARAR EL ARCHIVO REAL A EJECUTAR # ================================================================= if target_val == "web": js_file = None for f in os.listdir(tmp_dir): if f.endswith(".js"): js_file = os.path.join(tmp_dir, f) break if not js_file: await websocket.send_json({"type": "stdout", "data": "❌ Error: Compilación exitosa pero no se encontró un archivo .js en el directorio.\n"}) await websocket.send_json({"type": "exit", "data": "\n[Código 1]"}) await websocket.close() return cmd = ["node", js_file] elif target_val == "native": native_exe = None cand_linux = os.path.join(tmp_dir, "output") #cand_win = os.path.join(tmp_dir, "main.exe") if os.path.exists(cand_linux) and os.path.isfile(cand_linux): native_exe = cand_linux elif os.path.exists(cand_win) and os.path.isfile(cand_win): native_exe = cand_win else: archivos_ignorados = ('.tss', '.ll', '.o', '.obj', '.s', '.c', '.cpp', '.h') for f in os.listdir(tmp_dir): if not f.endswith(archivos_ignorados) and os.path.isfile(os.path.join(tmp_dir, f)): native_exe = os.path.join(tmp_dir, f) break if not native_exe: await websocket.send_json({"type": "stdout", "data": "❌ Error: Compilación exitosa pero no se detectó el binario final (solo archivos intermedios).\n"}) await websocket.send_json({"type": "exit", "data": "\n[Código 1]"}) await websocket.close() return st = os.stat(native_exe) os.chmod(native_exe, st.st_mode | stat.S_IEXEC) nombre_binario = os.path.basename(native_exe) cmd = ["stdbuf", "-i0", "-o0", "-e0", f"./{nombre_binario}"] else: # ================================================================= # MODO INTERP DIRECTO # ================================================================= cmd = [interpreter_path, source_file, syntax, "--target", "interp"] if is_debug: cmd.append("-d") # ===================================================================== # FASE 2: EJECUCIÓN INTERACTIVA ASÍNCRONA AL 100% # ===================================================================== print(f"Ejecutando WS {ws_id} (Target: {target_val}): {' '.join(cmd)}") # Cambiamos Popen por create_subprocess_exec para controlar el stdin/stdout sin deadlocks process = await asyncio.create_subprocess_exec( *cmd, stdin=asyncio.subprocess.PIPE, stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.STDOUT, cwd=tmp_dir ) async def read_output(): try: while True: # CORRECCIÓN VITAL: Leemos lo que haya al instante (hasta 1024 bytes). # Si Tesseract imprime "Ingresa algo: ", lo agarra de golpe y no espera el \n chunk = await process.stdout.read(1024) if not chunk: break await websocket.send_json({"type": "stdout", "data": chunk.decode("utf-8", errors="replace")}) await process.wait() return_code = process.returncode await websocket.send_json({"type": "exit", "data": f"\n[Ejecución finalizada con código {return_code}]"}) except Exception as e: print(f"Error en read_output: {e}") finally: try: await websocket.close() except: pass output_task = asyncio.create_task(read_output()) #
