// 5. Funciones (Variantes)
// ------------------------
// Función con tipo de retorno inferido (keyword 'function')
function sumar(int a, int b) {
    return a + b;
}

// Función con alias 'func' y retorno void (sin return)
func imprimirAlgo(string s) {
    print(s);
}

// Función con tipo explícito
int function multiplicar(int a, int b) {
    return a * b;
}

// Llamadas a funciones
var resultado = sumar(10,5);
print("valor #(sumar(10,5)) ");
imprimirAlgo("Ejecutando funcion void");

}

// FOR LOOP (Rango)

}

// SWITCH (Usa colon interno para casos, pero bloque externo con llaves)
switch (x) {
    case 1:

    default:

}
